public class ResourceManager

    // 加载文件
    private static string GetConfigFile(string fileNameWithExtension)
    {
        string url;
        #if UNITY_EDITOR || UNITY_STANDALONE
        	url = "file://" + Application.dataPath + "/streamingAssets/" + fileNameWithExtension;
        #elif UNITY_IPHONE
            url = "file://" + Application.dataPath + "/Raw/" + fileNameWithExtension;
        #elif UNITY_ANDROID
            url = "jar:file://" + Application.dataPath + "!/assets/" + fileNameWithExtension;
        #endif
        // WWW www = new WWW(url); // 弃用
        UnityWebRequest www = UnityWebRequest.Get(url);
        www.SendWebRequest();
        while (true)
        {
            if (www.downloadHandler.isDone)
                return www.downloadHandler.text;
        }
    }
    // 解析文件

        configMap = new Dictionary<string, string>();

        using (StringReader reader = new StringReader(fileContent)) // 字符串读取器

    public static T Load<T>(string prefabName) where T : Object
    {
        return Resources.Load<T>(configMap[prefabName]);
    }
}
