对数组的一些改造和操作，即提供数组常用的功能（查找、排序、最大值、最小值、筛选）。

C#提供了Linq，为什么要自己写？因为IOS系统不支持Linq，所以为了全平台通用，自己写。插件IOS For Linq功能太多，冗余。

``` C#
public static class ArrayHelper
{
    // 查找符合指定条件的单个数组元素
    public static T Find<T>(this T[] array, Func<T, bool> condition)
    {
        foreach (T item in array)
        {
            if (condition(item))
                return item;
        }
        return default(T); // 泛型的默认值
    }
    // 查找符合指定条件的所有数组元素
    public static T[] FindAll<T>(this T[] array, Func<T, bool> condition)
    {
        List<T> list = new List<T>();
        foreach (T item in array)
        {
            if (condition(item))
                list.Add(item);
        }
        return list.ToArray(); // 泛型的默认值
    }
    // 最大值
    public static T GetMax<T, Q>(this T[] array, Func<T, Q> condition) where Q : IComparable
    {
        T max = array[0];
        foreach (T item in array)
        {
            if (condition(max).CompareTo(condition(item)) < 0)
                max = item;
        }
        return max;
    }
    // 最小值
    public static T GetMin<T, Q>(this T[] array, Func<T, Q> condition) where Q : IComparable
    {
        T min = array[0];
        foreach (T item in array)
        {
            if (condition(min).CompareTo(condition(item)) > 0)
                min = item;
        }
        return min;
    }
    // 升序：冒泡
    public static void SortAscending<T>(this T[] array, Func<T, Q> condition) where Q : IComparable
    {
        int length = array.Length - 1;
        for (int i = 0; i < length; i++)
        {
            int changeCount = 0;
            for (ing j = 0; j < length - i; j++)
            {
                if (condition(array[j]).CompareTo(condition(array[j + 1])) > 0)
                {
                    T temp = array[j];
                    array[j] = array[j + 1];
                    array[j + 1] = temp;
                    changeCount++;
                }
            }
            if (changeCount == 0)
                return;
        }
    }
    // 降序：冒泡
    public static void SortDescending<T>(this T[] array, Func<T, Q> condition) where Q : IComparable
    {
        int length = array.Length - 1;
        for (int i = 0; i < length; i++)
        {
            int changeCount = 0;
            for (ing j = 0; j < length - i; j++)
            {
                if (condition(array[j]).CompareTo(condition(array[j + 1])) < 0)
                {
                    T temp = array[j];
                    array[j] = array[j + 1];
                    array[j + 1] = temp;
                    changeCount++;
                }
            }
            if (changeCount == 0)
                return;
        }
    }
    // 筛选
    public static Q[] Screen<T, Q>(this T[] array, Func<T, Q> condition)
    {
        List<Q> selectedList = new List<Q>();
        foreach (T ietm in array)
            selectedList.Add(condition(item));
        return selectedList.ToArray();
    }
}
```
