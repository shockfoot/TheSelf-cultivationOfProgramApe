在Unity中通常管理类写成单例。

``` C#
public calss XXManager : MonoBehaviour
{
    public static XXManager Instance { get; private set; }
    public void Awake()
    {
        Instance = this;
    }
}
```

此类写法当有多个不同的管理类时代码重复，且由于在`Awake()`中赋值，所以客户端代码只能在`Awake()`之后的脚本生命周期中访问。因此通常使用公共单例类来复用代码。

``` C#
// 脚本单例类
public class MonoSingleton<T> : MonoBehaviour where T : MonoSingleton<T> // T表示子类类型
{
    // 按需加载
    private static T instance;
    public static T Instance
    {
        get
        {
            if (instance == null)
            {
                // 在场景中根据类型查找引用
            	instance = FindObjectOfType<T>();
            	if (instace == null)
                	// 场景中无该脚本则自己添加(立即执行Awake)
                	new GameObject("SingletonOf" + typeof(T)).AddComponent<T>();
                else
                    instance.Init();
            }
            return instance;
        }
    }
    // 允许不在Awake之前被访问的管理类脚本自动赋值
    protected void Awake()
    {
        if (instance == null)
        {
            instance = this as T;
            Init();
        }
    }
    
    public virtual void Init() { }
}

public class XXManager : MonoSigleton<XXManager> { }
```
