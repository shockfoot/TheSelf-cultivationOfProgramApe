资源配置文件生成脚本GenerateResConfig：此脚本必须放在Editor目录下，并继承Editor，不会打包发布，可以在编译器中执行。

``` C#
public class GenerateResconfig : Editor
{
    private string resType = "prefab";
    private string[] fileFolders = { "Assets/Resources" };
    // 菜单项特性：该方法会在Unity中生成菜单
    [Menultem("Tools/Resources/Generate Resconfig File")]
    public static void Generate()
    {
        // 生成资源配置文件
        // 1、查找Resources目录下所有预制件的完整路径
        // FindAssets返回GUID
        string[] resFiles = 
            AssetDatabase.FindAssets($"t:{resType}", fileFolders);
        // 将GUID转化为路径
        for (int i = 0; i < resFiles.Length; i++)
        {
            resFiles[i] = AssetDatabase.GUIDToAssetPath(resFiles[i]);
            // 路径为：Assets/Resources/XXX/XXX.prefab
            // 2、生成映射关系：名称:路径
            string fileName = Path.GetFileNameWithoutExtension(resFiles[i]);
            string filePath = resFiles[i].Replace(fileFolders[0], string.Empty).
                Replace($".{resType}", string.Empty);
            resFiles[i] = fileName + ":" + filePath;
        }
        // 3、写入文件：为了兼容Android和IOS，只能放在StreamingAssets文件夹中
        // Unity特殊目录StreamingAssets文件夹：文件不会被压缩，可在移动端读取，PC端读写
        File.WriteAllLines("Assets/StreamingAssets/ComfigMap.txt", resFiles);
        AssetDatabase.Refresh(); // 刷新
    }
}
```
